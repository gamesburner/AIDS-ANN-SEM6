import numpy as np
import matplotlib.pyplot as plt


# Sigmoid Function
def sigmoid(x):
    return 1 / (1 + np.exp(-x))


# ReLU Function
def relu(x):
    return np.maximum(0, x)


# Tanh Function
def tanh(x):
    return np.tanh(x)


# Softmax Function
def softmax(x):
    return np.exp(x) / np.sum(np.exp(x))


# Create input values
x = np.linspace(-10, 10, 100)


# Plot Sigmoid
plt.plot(x, sigmoid(x))
plt.title("Sigmoid Function")
plt.xlabel("x")
plt.ylabel("sigmoid(x)")
plt.grid(True)
plt.show()


# Plot ReLU
plt.plot(x, relu(x))
plt.title("ReLU Function")
plt.xlabel("x")
plt.ylabel("relu(x)")
plt.grid(True)
plt.show()


# Plot Tanh
plt.plot(x, tanh(x))
plt.title("Tanh Function")
plt.xlabel("x")
plt.ylabel("tanh(x)")
plt.grid(True)
plt.show()


# Plot Softmax
y = softmax(x)

plt.plot(x, y)
plt.title("Softmax Function")
plt.xlabel("x")
plt.ylabel("softmax(x)")
plt.grid(True)
plt.show()
